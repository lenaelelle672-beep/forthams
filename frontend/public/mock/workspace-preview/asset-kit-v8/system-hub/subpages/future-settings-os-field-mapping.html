function transform(sourceValue, context) {
    // Format SAP functional location to hierarchical path
    if (!sourceValue) return "ROOT";

    const parts = sourceValue.split("-");
    if (parts.length < 2) return sourceValue;

    return `/` + parts.join('/');
}
