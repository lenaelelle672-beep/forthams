function onFieldLoad(context) {
    // Auto-calculate depreciation date based on acquisition date
    const aqDate = context.getValue('acquisition_date');
    const lifespan = context.getMeta('expected_lifespan_years');

    if (aqDate && lifespan) {
        let end = new Date(aqDate);
        end.setFullYear(end.getFullYear() + lifespan);
        context.setValue('depreciation_end_date', end.toISOString());

        // Lock field if calculated automatically
        context.setReadOnly('depreciation_end_date', true);
    }
}
